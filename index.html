// JAMNOISEIR VNK #include #include #define IR_TX_PIN //TỰ THÊM IRsend irsend(IR_TX_PIN); const uint32_t JAMMER_CODE = 0xFFFFFFFF; const uint32_t JAMMER_INTERVAL_MS = 80; unsigned long lastSend = 0; bool jammerEnabled = true; void setup() { Serial.begin(115200); delay(500); irsend.begin(); Serial.println("IR Jammer started"); } void loop() { if (Serial.available()) { char cmd = Serial.read(); cmd = tolower(cmd); if (cmd == '1') { jammerEnabled = true; Serial.println("Jammer: ON"); } else if (cmd == '0') { jammerEnabled = false; Serial.println("Jammer: OFF"); } while (Serial.available()) { Serial.read(); } } if (jammerEnabled && (millis() - lastSend >= JAMMER_INTERVAL_MS)) { lastSend = millis(); irsend.sendNEC(JAMMER_CODE, 32); } delay(1); }
